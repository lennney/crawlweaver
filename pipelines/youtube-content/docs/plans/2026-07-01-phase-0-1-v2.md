```python
"""数据库模型 — YouTube Content Factory"""
from datetime import datetime, timezone
from sqlalchemy import (
    create_engine, Column, String, Integer, Float,
    DateTime, JSON, Text, Enum, ForeignKey
)
from sqlalchemy.orm import DeclarativeBase, sessionmaker
import enum


class Base(DeclarativeBase):
    pass


class PipelineStage(str, enum.Enum):
    IDEA = "idea"
    BRIEFED = "briefed"
    PRODUCING = "producing"
    PRODUCED = "produced"
    SEO_DONE = "seo_done"
    REVIEWING = "reviewing"
    APPROVED = "approved"
    PUBLISHING = "publishing"
    LIVE = "live"
    FAILED = "failed"


class TrendingVideo(Base):
    """每日热门视频（YouTube + TikTok）"""
    __tablename__ = "trending_videos"

    id = Column(String, primary_key=True)          # platform:video_id
    platform = Column(String, nullable=False)       # youtube / tiktok
    title = Column(String, nullable=False)
    url = Column(String)
    views = Column(Integer, default=0)
    likes = Column(Integer, default=0)
    comments = Column(Integer, default=0)
    shares = Column(Integer, default=0)
    duration = Column(Integer)                      # 秒
    tags = Column(JSON, default=list)
    description = Column(Text)
    thumbnail_url = Column(String)
    published_at = Column(DateTime, nullable=True)  # 采集时可能未知
    fetched_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))
    trend_score = Column(Float, default=0.0)


class Video(Base):
    """生产的视频"""
    __tablename__ = "videos"

    id = Column(String, primary_key=True)
    stage = Column(
        Enum(PipelineStage),
        nullable=False,
        default=PipelineStage.IDEA
    )
    title = Column(String)
    hook_angle = Column(String)
    source_trend_id = Column(String, ForeignKey("trending_videos.id"))
    platform = Column(String, nullable=False)

    # 素材
    material_path = Column(String)
    output_path = Column(String)

    # SEO
    seo_package = Column(JSON)

    # 审核
    review_status = Column(String, nullable=True)  # pending/approved/rejected
    review_notes = Column(Text, nullable=True)

    # 发布
    youtube_id = Column(String)
    scheduled_publish_at = Column(DateTime, nullable=True)

    # 分析（Phase 2+）
    analytics = Column(JSON, nullable=True)

    # 元数据
    crew_run_id = Column(String, nullable=True)
    created_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))
    updated_at = Column(DateTime, onupdate=lambda: datetime.now(timezone.utc))
    error = Column(Text)


def get_engine(db_url: str):
    return create_engine(db_url, pool_size=5, max_overflow=10)


def get_session(engine):
    return sessionmaker(bind=engine)()
```

### Task 1.1: 热度分算法

| 项 | 值 |
|---|-----|
| **新建文件** | `trending_aggregator/scorer.py` |
| **代码** | ~40 行 |

```python
"""热度分算法 — 含时间衰减"""
import math


def compute_trend_score(
    views: int,
    likes: int,
    comments: int,
    shares: int,
    published_days_ago: float = 0,
) -> float:
    """
    计算热度分 (0-100)
    
    - views: 百万播放 = 满分
    - likes/views 比率: 上限 20%
    - comments/views 比率: 上限 5%
    - shares/views 比率: 上限 10%
    - 时间衰减: 24h 内无衰减，之后 7 天半衰期
    """
    views = max(views, 1)
    
    # 基础分
    views_score = min(views / 1_000_000, 1.0) * 40
    likes_rate = min(likes / views, 0.20)
    likes_score = (likes_rate / 0.20) * 25
    comments_rate = min(comments / views, 0.05)
    comments_score = (comments_rate / 0.05) * 15
    shares_rate = min(shares / views, 0.10)
    shares_score = (shares_rate / 0.10) * 20
    
    raw = views_score + likes_score + comments_score + shares_score
    
    # 时间衰减：24h 内无衰减，之后 7 天半衰期
    if published_days_ago > 1:
        decay = 2.0 ** (-(published_days_ago - 1) / 7)
        raw *= decay
    
    return round(raw, 1)
```

```python
"""YouTube Trending 采集器 — 用 yt-dlp 抓取（无需 API key）"""
import json
import logging
import os
import subprocess
from datetime import datetime, timezone
from sqlalchemy.orm import Session

from factory.models import TrendingVideo, get_engine, get_session
from trending_aggregator.scorer import compute_trend_score

logger = logging.getLogger(__name__)


def scrape_youtube_trending(gl: str = "US", limit: int = 50) -> list[dict]:
    """
    用 yt-dlp 抓取 YouTube Trending
    
    依赖: yt-dlp + ffmpeg (系统级)
    """
    cmd = [
        "yt-dlp",
        f"https://www.youtube.com/feed/trending?gl={gl}",
        "--dump-json",
        "--flat-playlist",
        "--playlist-end", str(limit),
        "--no-warnings",
        "--ignore-errors",
    ]
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
    except subprocess.TimeoutExpired:
        logger.error("YouTube Trending 抓取超时")
        return []
    
    if result.returncode != 0:
        logger.error(f"yt-dlp 失败: {result.stderr[:200]}")
        return []
    
    videos = []
    for line in result.stdout.strip().split("\n"):
        if not line:
            continue
        try:
            data = json.loads(line)
            videos.append({
                "id": f"youtube:{data['id']}",
                "title": data.get("title", ""),
                "url": data.get("webpage_url", ""),
                "views": data.get("view_count") or 0,
                "likes": data.get("like_count") or 0,
                "comments": data.get("comment_count") or 0,
                "shares": 0,  # yt-dlp 不返回 shares
                "duration": data.get("duration", 0),
                "tags": data.get("tags") or [],
                "description": data.get("description", ""),
                "thumbnail_url": data.get("thumbnail", ""),
                "published_at": _parse_upload_date(data.get("upload_date")),
            })
        except (json.JSONDecodeError, KeyError) as e:
            logger.warning(f"跳过损坏行: {e}")
    
    return videos


def _parse_upload_date(date_str: str | None) -> datetime | None:
    """yt-dlp 的 upload_date 格式: YYYYMMDD"""
    if not date_str:
        return None
    try:
        return datetime.strptime(date_str, "%Y%m%d").replace(tzinfo=timezone.utc)
    except ValueError:
        return None


def save_to_db(session: Session, videos: list[dict], platform: str) -> int:
    """保存到 trending_videos，返回新增/更新数"""
    now = datetime.now(timezone.utc)
    count = 0
    for v in videos:
        record = session.get(TrendingVideo, v["id"])
        
        # 计算热度分（含时间衰减）
        if v["published_at"]:
            days_ago = (now - v["published_at"]).total_seconds() / 86400
        else:
            days_ago = 0
        
        score = compute_trend_score(
            views=v["views"],
            likes=v["likes"],
            comments=v["comments"],
            shares=v.get("shares", 0),
            published_days_ago=days_ago,
        )
        
        if record:
            record.views = v["views"]
            record.likes = v["likes"]
            record.comments = v["comments"]
            record.trend_score = score
        else:
            session.add(TrendingVideo(
                id=v["id"],
                platform=platform,
                title=v["title"],
                url=v["url"],
                views=v["views"],
                likes=v["likes"],
                comments=v["comments"],
                duration=v["duration"],
                tags=v["tags"],
                description=v["description"],
                thumbnail_url=v["thumbnail_url"],
                published_at=v["published_at"],
                trend_score=score,
            ))
        count += 1
    session.commit()
    return count


def run():
    """入口：采集 YouTube Trending → 入库"""
    logging.basicConfig(level=logging.INFO)
    
    db_url = os.environ["DATABASE_URL"]
    engine = get_engine(db_url)
    session = get_session(engine)
    
    gl = os.getenv("YOUTUBE_TRENDING_GL", "US")
    limit = int(os.getenv("YOUTUBE_TRENDING_LIMIT", "50"))
    
    logger.info(f"开始采集 YouTube Trending (gl={gl}, limit={limit})...")
    videos = scrape_youtube_trending(gl=gl, limit=limit)
    
    if not videos:
        logger.warning("未获取到任何视频，检查网络或 yt-dlp")
        return
    
    count = save_to_db(session, videos, platform="youtube")
    logger.info(f"完成: {count} 个视频入库")


if __name__ == "__main__":
    run()
```

```python
    engine = get_engine(db_url)
    session = get_session(engine)
    
    region = os.getenv("TIKTOK_REGION", "US")
    limit = int(os.getenv("TIKTOK_LIMIT", "30"))
    
    logger.info(f"开始采集 TikTok Trending (region={region}, limit={limit})...")
    videos = scrape_tiktok_trending(region=region, limit=limit)
    
    if not videos:
        logger.warning("TikTok 未获取到视频（可能需要代理）")

    count = save_to_db(session, videos)
```
