```python
from sqlalchemy import create_engine, Column, String, Integer, Float, DateTime, JSON, Text
from sqlalchemy.orm import DeclarativeBase, sessionmaker
from datetime import datetime, timezone

class Base(DeclarativeBase):
    pass

class TrendingVideo(Base):
    __tablename__ = "trending_videos"
    
    id = Column(String, primary_key=True)
    platform = Column(String, nullable=False)      # youtube / tiktok / instagram
    title = Column(String, nullable=False)
    url = Column(String)
    views = Column(Integer, default=0)
    likes = Column(Integer, default=0)
    comments = Column(Integer, default=0)
    duration = Column(Integer)                      # 秒
    tags = Column(JSON, default=list)
    description = Column(Text)
    thumbnail_url = Column(String)
    published_at = Column(DateTime)
    fetched_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))
    trend_score = Column(Float, default=0.0)        # 计算的热度分

class Video(Base):
    __tablename__ = "videos"
    
    id = Column(String, primary_key=True)
    stage = Column(String, nullable=False, default="idea")  # PipelineStage
    title = Column(String)
    hook_angle = Column(String)
    source_trend_id = Column(String)                # FK → trending_videos.id
    material_path = Column(String)
    output_path = Column(String)
    seo_package = Column(JSON)
    youtube_id = Column(String)
    created_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))
    updated_at = Column(DateTime, onupdate=lambda: datetime.now(timezone.utc))
    error = Column(Text)

def get_engine(db_url: str):
    return create_engine(db_url)

def get_session(engine):
    return sessionmaker(bind=engine)()
```

## Phase 1: Trending Scout

### Task 1.1: YouTube Trending 采集器

| 项 | 值 |
|---|-----|
| **文件** | `trending_aggregator/youtube_scout.py` |
| **代码** | ~120 行 |

```python
"""
YouTube Trending 采集器
每天抓取 YouTube Trending 页面的热门视频

数据来源：YouTube Trending 页面（无需 API key）
工具：yt-dlp
"""

import json
import subprocess
from datetime import datetime, timezone
from sqlalchemy.orm import Session
from factory.models import TrendingVideo


def scrape_youtube_trending(country: str = "US", limit: int = 50) -> list[dict]:
    """
    用 yt-dlp 抓取 YouTube Trending 页面
    
    yt-dlp 支持 --dump-json 输出结构化数据，
    包含 title, view_count, like_count, duration, tags 等
    """
    cmd = [
        "yt-dlp",
        f"https://www.youtube.com/feed/trending?gl={country}",
        "--dump-json",
        "--flat-playlist",
        "--playlist-end", str(limit),
        "--no-warnings",
    ]
    
    result = subprocess.run(cmd, capture_output=True, text=True, timeout=120)
    
    videos = []
    for line in result.stdout.strip().split("\n"):
        if not line:
            continue
        data = json.loads(line)
        videos.append({
            "id": data["id"],
            "title": data.get("title", ""),
            "url": data.get("webpage_url", ""),
            "views": data.get("view_count", 0),
            "likes": data.get("like_count", 0),
            "comments": data.get("comment_count", 0),
            "duration": data.get("duration", 0),
            "tags": data.get("tags", []),
            "description": data.get("description", ""),
            "thumbnail_url": data.get("thumbnail", ""),
            "published_at": None,  # trending 页面不返回准确时间
        })
    
    return videos


def compute_trend_score(video: dict) -> float:
    """
    计算热度分（0-100）
    
    算法：views * 0.5 + likes_rate * 30 + comments_rate * 20
    后续会加入时间衰减因子
    """
    views = max(video["views"], 1)
    likes_rate = min(video["likes"] / views, 0.2)  # cap at 20%
    comments_rate = min(video["comments"] / views, 0.05)
    
    views_score = min(views / 1_000_000, 1.0) * 50  # 100万 views = 满分
    likes_score = (likes_rate / 0.2) * 30
    comments_score = (comments_rate / 0.05) * 20
    
    return round(views_score + likes_score + comments_score, 1)


def save_to_db(session: Session, videos: list[dict], platform: str = "youtube"):
    """保存到 trending_videos 表，重复的更新"""
    for v in videos:
        record = session.get(TrendingVideo, v["id"])
        if record:
            # 更新已有记录
            record.views = v["views"]
            record.likes = v["likes"]
            record.comments = v["comments"]
            record.trend_score = compute_trend_score(v)
        else:
            session.add(TrendingVideo(
                id=v["id"],
                platform=platform,
                title=v["title"],
                url=v["url"],
                views=v["views"],
                likes=v["likes"],
                comments=v["comments"],
                duration=v["duration"],
                tags=v["tags"],
                description=v["description"],
                thumbnail_url=v["thumbnail_url"],
                trend_score=compute_trend_score(v),
            ))
    session.commit()


def run():
    """每日执行入口"""
    from factory.models import get_engine, get_session
    import os
    
    db_url = os.getenv("DATABASE_URL", "postgresql://ytfactory:test@localhost:5432/ytfactory")
    engine = get_engine(db_url)
    session = get_session(engine)
    
    print(f"[{datetime.now(timezone.utc)}] 开始采集 YouTube Trending...")
    videos = scrape_youtube_trending(country="US", limit=50)
    save_to_db(session, videos)
    print(f"[{datetime.now(timezone.utc)}] 保存完成：{len(videos)} 个视频")
```

```python
import pytest
from trending_aggregator.youtube_scout import compute_trend_score, save_to_db
from factory.models import TrendingVideo, Base, get_engine, get_session


class TestTrendScore:
    def test_zero_views(self):
        video = {"views": 0, "likes": 0, "comments": 0}
        # views 至少为 1 避免除零
        assert compute_trend_score({"views": 1, "likes": 0, "comments": 0}) >= 0
    
    def test_viral_video(self):
        video = {"views": 5_000_000, "likes": 500_000, "comments": 50_000}
        score = compute_trend_score(video)
        assert 80 <= score <= 100
    
    def test_normal_video(self):
        video = {"views": 100_000, "likes": 5_000, "comments": 1_000}
        score = compute_trend_score(video)
        assert 20 <= score <= 60


class TestDatabase:
    @pytest.fixture
    def session(self):
        engine = get_engine("sqlite:///:memory:")
        Base.metadata.create_all(engine)
        return get_session(engine)
    
    def test_save_new_video(self, session):
        videos = [{
            "id": "test123",
            "title": "Test Video",
            "url": "https://youtube.com/watch?v=test123",
            "views": 10000, "likes": 500, "comments": 100,
            "duration": 600,
            "tags": ["test"],
            "description": "A test video",
            "thumbnail_url": "https://img.youtube.com/vi/test123/default.jpg",
        }]
        save_to_db(session, videos, platform="youtube")
        
        record = session.get(TrendingVideo, "test123")
        assert record is not None
        assert record.title == "Test Video"
        assert record.trend_score > 0
    
    def test_update_existing_video(self, session):
        videos = [{"id": "test456", "title": "Old Title",
                    "url": "...", "views": 1000, "likes": 10, "comments": 1,
                    "duration": 300, "tags": [], "description": "", 
                    "thumbnail_url": ""}]
        save_to_db(session, videos, platform="youtube")
        
        # 模拟再次采集（views 增长）
        videos[0]["views"] = 5000
        videos[0]["likes"] = 50
        save_to_db(session, videos, platform="youtube")
        
        record = session.get(TrendingVideo, "test456")
        assert record.views == 5000
```
